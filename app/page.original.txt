"use client";

import { useState } from "react";

const fieldClass =
  "mb-2 w-full rounded-lg border border-slate-300 px-3 py-2.5 focus:outline-2 focus:outline-offset-1 focus:outline-blue-600";

export default function Home() {
  const [status, setStatus] = useState("");

  function handleSubmit(event) {
    event.preventDefault();
    setStatus("Submission is not set up yet.");
  }

  return (
    <main className="grid min-h-screen place-items-center bg-slate-50 p-6 text-slate-900">
      <form
        className="grid w-full max-w-md gap-2 rounded-xl border border-slate-200 bg-white p-8 shadow-lg shadow-slate-900/5"
        onSubmit={handleSubmit}
      >
        <h1 className="mb-3 text-3xl font-bold">Get in touch</h1>

        <label className="text-sm font-semibold" htmlFor="name">Name</label>
        <input className={fieldClass} id="name" name="name" type="text" autoComplete="name" required />

        <label className="text-sm font-semibold" htmlFor="email">Email</label>
        <input className={fieldClass} id="email" name="email" type="email" autoComplete="email" required />

        <label className="text-sm font-semibold" htmlFor="message">Message</label>
        <textarea className={fieldClass} id="message" name="message" rows="5" required />

        <button
          className="cursor-pointer rounded-lg bg-blue-600 px-3 py-3 font-semibold text-white hover:bg-blue-700"
          type="submit"
        >
          Submit
        </button>
        {status && <p className="text-sm text-slate-600" role="status">{status}</p>}
      </form>
    </main>
  );
}
